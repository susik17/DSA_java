#--------------from collections import OrderedDict

# Create an OrderedDict
od = OrderedDict()
od['a'] = 1
od['b'] = 2
od['c'] = 3

# Move 'a' to the very end
od.move_to_end('a')
print(od)  # Outputs: OrderedDict([('b', 2), ('c', 3), ('a', 1)])

# Pop the very first item (FIFO)
first_item = od.popitem(last=False)
print(first_item)  # Outputs: ('b', 2)
